    // Background Animation
    const animations = document.querySelectorAll('.bg-animation');
    animations.forEach(container => {
      const snippets = container.dataset.snippets.split(',');
      function createSnippet() {
        const span = document.createElement('span');
        span.textContent = snippets[Math.floor(Math.random() * snippets.length)];
        span.style.left = Math.random() * 100 + 'vw';
        span.style.animationDuration = (Math.random() * 6 + 12) + 's';
        span.style.animationDelay = (Math.random() * 3) + 's';
        container.appendChild(span);
        setTimeout(() => span.remove(), 20000);
      }
      for (let i = 0; i < 10; i++) createSnippet(); // Initial burst
      setInterval(createSnippet, 1200);
    });

    // Typewriter Animation
    const words = ['Design', 'Develop', 'Innovate'];
    let wordIndex = 0;
    let charIndex = 0;
    let isDeleting = false;
    const typewriterElement = document.getElementById('typewriter');

    function type() {
      const currentWord = words[wordIndex];
      if (isDeleting) {
        typewriterElement.textContent = currentWord.substring(0, charIndex--);
        if (charIndex < 0) {
          isDeleting = false;
          wordIndex = (wordIndex + 1) % words.length;
          setTimeout(type, 500);
        } else {
          setTimeout(type, 100);
        }
      } else {
        typewriterElement.textContent = currentWord.substring(0, charIndex++);
        if (charIndex > currentWord.length) {
          isDeleting = true;
          setTimeout(type, 1000);
        } else {
          setTimeout(type, 150);
        }
      }
    }
    type();
